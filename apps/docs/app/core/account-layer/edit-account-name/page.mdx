# editAccountName

Rename a subaccount.

```ts
import { editAccountName } from "@symmio/trading-core";

const hash = await editAccountName(config, {
  account: "0xsub...",
  name: "Trading bot",
});
```

## Parameters

<Properties>
  <Property name="account" type="Address" required>
    The subaccount being renamed. The signer must equal the subaccount's `owner` on chain.
  </Property>
  <Property name="name" type="string" required>
    New display name.
  </Property>
  <Property name="chainId" type="number" optional>
    Optional chain override.
  </Property>
  <Property name="gasless" type="boolean | GaslessWriteOptions" optional>
    Relay this write through the [gasless execution mode](/core/gasless/relayable-writes) instead of the wallet. A
    boolean for the common case, or a per-call `GaslessWriteOptions` bag; omitted, the chain's `execution.mode` decides.
  </Property>
</Properties>

## Returns

<Returns type="Hash">The broadcast tx hash.</Returns>
